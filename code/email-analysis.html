datasets <- unique(data_e$file_origin)

if (length(datasets) == 0) {
  stop("`datasets` is empty — check that `data_e$file_origin` is the correct column name.")
}

for (i in seq_along(datasets)) {
  ds <- datasets[i]
  message("Word-level sentiment: ", ds, " (", i, "/", length(datasets), ")")
  
  ds_safe <- str_replace_all(ds, "[^A-Za-z0-9_-]", "_")
  doc_sent_file <- file.path(doc_sent_dir, paste0(ds_safe, ".csv"))
  word_cnt_file <- file.path(word_cnt_dir, paste0(ds_safe, ".csv"))
  
  if (file.exists(doc_sent_file) && file.exists(word_cnt_file)) {
    message("  already done, skipping")
    next
  }
  
  chunk_tokens <- data_e |>
    filter(file_origin == ds) |>
    select(all_of(c(ID_COL, "file_origin", TEXT_COL))) |>
    mutate(!!TEXT_COL := str_remove_all(!!sym(TEXT_COL), redaction_pattern)) |>
    unnest_tokens(word, !!sym(TEXT_COL)) |>
    mutate(word = lemmatize_words(word))
  
  doc_sent <- chunk_tokens |>
    inner_join(bing, by = "word", relationship = "many-to-many") |>
    count(!!sym(ID_COL), file_origin, sentiment) |>
    pivot_wider(names_from = sentiment, values_from = n, values_fill = 0)
  
  if (!"positive" %in% names(doc_sent)) doc_sent$positive <- 0
  if (!"negative" %in% names(doc_sent)) doc_sent$negative <- 0
  doc_sent <- doc_sent |> mutate(sentiment = positive - negative)
  
  write.csv(doc_sent, doc_sent_file, row.names = FALSE)
  
  chunk_word_counts <- chunk_tokens |>
    inner_join(bing, by = "word", relationship = "many-to-many") |>
    count(word, sentiment, sort = FALSE)
  
  write.csv(chunk_word_counts, word_cnt_file, row.names = FALSE)
  
  rm(chunk_tokens, doc_sent, chunk_word_counts)
  gc()
}

Comparing Lexicons across dataset 11

chunk_size <- 200  # same as your dataset-level plot, for consistency

# ============================================================================
# Tokenize + lemmatize data-set-11-files once, reuse for all three lexicons
# ============================================================================

ds11_tokens <- data_e |>
  filter(file_origin == "data-set-11-files") |>
  select(all_of(c(ID_COL, "file_origin", TEXT_COL))) |>
  arrange(!!sym(ID_COL)) |>
  mutate(doc_index = (row_number() - 1) %/% chunk_size) |>  # assign chunk before unnesting
  mutate(!!TEXT_COL := str_remove_all(!!sym(TEXT_COL), redaction_pattern)) |>
  unnest_tokens(word, !!sym(TEXT_COL)) |>
  mutate(word = lemmatize_words(word))

# ============================================================================
# AFINN — sum of values per chunk
# ============================================================================

afinn_ds11 <- ds11_tokens |>
  inner_join(afinn, by = "word") |>
  group_by(index = doc_index) |>
  summarise(sentiment = sum(value), .groups = "drop") |>
  mutate(method = "AFINN")

# ============================================================================
# Bing + NRC — positive/negative counts per chunk
# ============================================================================

bing_and_nrc_ds11 <- bind_rows(
  ds11_tokens |>
    inner_join(bing, by = "word", relationship = "many-to-many") |>
    mutate(method = "Bing et al."),
  ds11_tokens |>
    inner_join(
      nrc |> filter(sentiment %in% c("positive", "negative")),
      by = "word", relationship = "many-to-many"
    ) |>
    mutate(method = "NRC")
) |>
  count(method, index = doc_index, sentiment) |>
  pivot_wider(names_from = sentiment, values_from = n, values_fill = 0) |>
  mutate(sentiment = positive - negative)

# ============================================================================
# Combine and plot
# ============================================================================

bind_rows(afinn_ds11, bing_and_nrc_ds11) |>
  ggplot(aes(index, sentiment, fill = method)) +
  geom_col(show.legend = FALSE) +
  facet_wrap(~method, ncol = 1, scales = "free_y") +
